cmake_minimum_required(VERSION 3.21)
project(amoled_studio C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
FetchContent_Declare(lvgl URL https://codeload.github.com/lvgl/lvgl/tar.gz/c033a98afddd65aaafeebea625382a94020fe4a7 URL_HASH SHA256=9c6f8230d0b0e11141aad8f047af699322dbd3d37b2c41170b1526c12d5dfbb3 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(lvgl)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_library(studio_board src/board/board.c)
target_include_directories(studio_board PUBLIC src/board)
add_library(studio_mock src/hal/mock_platform.c)
target_include_directories(studio_mock PUBLIC src/hal)
add_library(studio_app src/app/app.c)
target_include_directories(studio_app PUBLIC src/app)
target_link_libraries(studio_app PUBLIC lvgl)
target_include_directories(studio_app PUBLIC src/hal)
add_library(studio_circle src/platform/circle.c)
target_include_directories(studio_circle PUBLIC src/platform)
add_executable(amoled-simulator src/platform/sdl_main.c)
target_link_libraries(amoled-simulator PRIVATE studio_app studio_circle studio_board studio_mock PkgConfig::SDL2)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_circle tests/test_circle.c)
  target_link_libraries(test_circle PRIVATE studio_circle)
  add_executable(test_platform tests/test_platform.c)
  target_link_libraries(test_platform PRIVATE studio_board studio_mock)
  target_compile_options(test_circle PRIVATE -UNDEBUG)
  target_compile_options(test_platform PRIVATE -UNDEBUG)
  add_test(NAME platform COMMAND test_platform)
  add_test(NAME circle COMMAND test_circle)
  add_test(NAME simulator_smoke COMMAND amoled-simulator --smoke)
  set_tests_properties(simulator_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 15)
endif()

  add_test(NAME simulator_mock_configuration COMMAND amoled-simulator --smoke)
